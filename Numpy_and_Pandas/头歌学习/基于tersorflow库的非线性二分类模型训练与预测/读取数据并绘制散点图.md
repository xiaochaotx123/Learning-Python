```python
import pandas as pd  

import matplotlib.pyplot as plt

data = pd.read_csv('data.csv')  


passed = plt.scatter(data.loc[:, 'x1'][data['y']==1], data.loc[:, 'x2'][data['y']==1])
failed = plt.scatter(data.loc[:, 'x1'][data['y']==0], data.loc[:, 'x2'][data['y']==0])

plt.legend(handles=[passed, failed], labels=['passed', 'failed'], loc='upper right')

plt.xlabel('x1')
plt.ylabel('x2')

plt.title('raw data')

plt.savefig('step1/result/tu.png')

```
