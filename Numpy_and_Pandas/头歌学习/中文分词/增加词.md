```python
import jieba
import jieba.posseg as pseg

# 动态添加可能未被正确切分的词
jieba.add_word("大模型", tag='n')
jieba.add_word("绝绝子", tag='a')

text = input().strip()

# 分词并进行词性标注，筛选名词和形容词
for word, flag in pseg.cut(text):
    if flag.startswith('n') or flag.startswith('a'):
        print(word, end='\t')

print()
```
